Executive Summary

MYSTES is a flight price arbitrage platform that exploits geographic pricing disparities in airline ticket markets. Airlines display different prices for identical flights based on the user's geographic location, currency, and market. This platform automates the discovery of these price differences, enabling users to save 15-70% on international flights.

Core Innovation: A hybrid search architecture that combines residential proxy web scraping with API data aggregation, running parallel searches to deliver both accurate regional pricing AND complete flight details in a single user query.

# --- CURRENCY CONVERSION ---
CURRENCY_RATES_TO_USD = {
    "USD": 1.0,
    "EUR": 1.08,    # 1 EUR ≈ 1.08 USD
    "GBP": 1.26,    # 1 GBP ≈ 1.26 USD
    "JPY": 0.0067,  # 1 JPY ≈ 0.0067 USD
}

def convert_price(price, from_currency, to_currency="USD"):
    """Convert price from one currency to another."""
    usd_rate = CURRENCY_RATES_TO_USD.get(from_currency, 1.0)
    target_rate = CURRENCY_RATES_TO_USD.get(to_currency, 1.0)
    price_in_usd = float(price) * usd_rate
    return price_in_usd / target_rate


# --- DEAL CALCULATOR ---
def calculate_deal(home_price, arbitrage_price, airline, market):
    """
    Calculate deal breakdown with platform fee.

    Example:
      US Price: $451, Spain Price: $350
      Gross Savings: $101
      User Savings (75%): $75.75
      Platform Fee (25%): $25.25
    """
    gross_savings = home_price - arbitrage_price
    if gross_savings <= 0:
        return None

    platform_fee = gross_savings * 0.25  # 25% cut
    platform_fee = max(platform_fee, 3.00)   # Min $3
    platform_fee = min(platform_fee, 50.00)  # Max $50

    user_savings = gross_savings - platform_fee

    return {
        "home_price": round(home_price, 2),
        "arbitrage_price": round(arbitrage_price, 2),
        "gross_savings": round(gross_savings, 2),
        "platform_fee_usd": round(platform_fee, 2),
        "user_savings": round(user_savings, 2),
        "user_saves_pct": round((user_savings / home_price) * 100, 1),
        "is_good_deal": user_savings >= 10.00,
    }


# --- XRPL PAYMENT ---
def generate_payment_request(deal_id, fee_usd):
    """Generate XRP payment request for platform fee."""
    xrp_amount = fee_usd / XRPL_CONFIG["xrp_usd_rate"]
    destination_tag = abs(hash(deal_id)) % 2147483647

    return {
        "destination": XRPL_CONFIG["platform_wallet_address"],
        "destination_tag": destination_tag,
        "amount_xrp": round(xrp_amount, 6),
        "amount_usd": fee_usd,
    }

class Itinerary:
    """
    Multi-flight travel itinerary builder.

    Example: LAX → HND → BKK → SIN → SYD
    Calculates total savings across all legs.
    """

    def __init__(self, name: str = "My Trip"):
        self.name = name
        self.legs = []

    def add_leg(self, origin, destination, date):
        """Add a flight leg to the itinerary."""
        self.legs.append({
            "origin": origin,
            "destination": destination,
            "date": date
        })

    def search_all_legs(self):
        """Search for deals on all legs."""
        total_savings = 0

        for leg in self.legs:
            result = search_global(
                leg["origin"],
                leg["destination"],
                leg["date"]
            )
            if result.get("deals"):
                total_savings += result["deals"][0]["user_savings"]

        # 10% discount on fees for 3+ legs
        if len(self.legs) >= 3:
            total_savings *= 1.10

        return {"total_savings": total_savings}


def search_multi_city(legs):
    """
    Search multi-city itinerary.

    Example:
        search_multi_city([
            ("LAX", "HND", "2026-03-01"),
            ("HND", "BKK", "2026-03-05"),
            ("BKK", "SYD", "2026-03-10"),
        ])
    """
    itinerary = Itinerary()
    for origin, dest, date in legs:
        itinerary.add_leg(origin, dest, date)
    return itinerary.search_all_legs()

import asyncio
import re
from playwright.async_api import async_playwright
from proxy_manager import get_proxy_for_market

MARKET_CONFIG = {
    "US": {"domain": "google.com", "currency": "USD"},
    "ES": {"domain": "google.es", "currency": "EUR"},
    "UK": {"domain": "google.co.uk", "currency": "GBP"},
    "DE": {"domain": "google.de", "currency": "EUR"},
}


async def scrape_flights_from_market(origin, destination, date, market):
    """
    Scrape Google Flights from a specific market via proxy.

    Returns real regional prices as seen by users in that country.
    """
    proxy_config = get_proxy_for_market(market)
    config = MARKET_CONFIG[market]

    url = f"https://www.{config['domain']}/travel/flights?" \
          f"q=Flights+from+{origin}+to+{destination}+on+{date}"

    async with async_playwright() as p:
        browser = await p.chromium.launch(
            headless=True,
            proxy=proxy_config
        )

        context = await browser.new_context(
            viewport={"width": 1920, "height": 1080},
            user_agent="Mozilla/5.0 (Windows NT 10.0...",
        )

        page = await context.new_page()
        await page.goto(url, wait_until="networkidle")

        # Handle EU cookie consent
        consent_btns = [
            'button:has-text("Accept all")',
            'button:has-text("Aceptar todo")',  # Spanish
            'button:has-text("Alle akzeptieren")',  # German
        ]
        for selector in consent_btns:
            try:
                btn = page.locator(selector).first
                if await btn.count() > 0:
                    await btn.click()
                    break
            except:
                pass

        # Extract prices
        page_text = await page.inner_text('body')
        flights = extract_prices(page_text, market, config["currency"])

        await browser.close()
        return flights


def extract_prices(text, market, currency):
    """Extract flight prices using currency-specific regex."""
    patterns = {
        "EUR": [r'(\d{3,4})\s*€', r'€\s*(\d{3,4})'],
        "USD": [r'\$\s*(\d{3,4})'],
        "GBP": [r'£\s*(\d{3,4})'],
    }

    flights = []
    for pattern in patterns.get(currency, []):
        for match in re.findall(pattern, text):
            price = int(match.replace(",", ""))
            if 100 < price < 5000:
                flights.append({
                    "price": float(price),
                    "currency": currency,
                    "market": market
                })

    return sorted(flights, key=lambda x: x["price"])[:15]

# proxy_manager.py - Proxy Management

import os
from typing import Dict, Optional

MARKET_TO_COUNTRY = {
    "US": "US", "JP": "JP", "UK": "GB",
    "DE": "DE", "FR": "FR", "ES": "ES",
    "IT": "IT", "AU": "AU", "CA": "CA",
    # ... 40+ countries
}


class ProxyManager:
    """
    Manages residential proxy connections.

    Supports Webshare.io with automatic country targeting.
    Format: username-country-XX:password@proxy.webshare.io:80
    """

    def __init__(self):
        self.username = os.getenv("WEBSHARE_USERNAME", "")
        self.password = os.getenv("WEBSHARE_PASSWORD", "")
        self.host = "proxy.webshare.io"
        self.port = "80"

    def get_proxy(self, market: str) -> Optional[Dict]:
        """Get proxy config for a specific market."""
        if not self.username:
            return None

        country = MARKET_TO_COUNTRY.get(market, market)
        username = f"{self.username}-country-{country}"
        url = f"http://{username}:{self.password}@{self.host}:{self.port}"

        return {"http": url, "https": url}

    def test_proxy(self, market: str) -> dict:
        """Test proxy connection and verify location."""
        import requests
        proxy = self.get_proxy(market)

        response = requests.get(
            "https://ipinfo.io/json",
            proxies=proxy,
            timeout=15
        )
        data = response.json()

        return {
            "success": True,
            "detected_country": data.get("country"),
            "ip": data.get("ip"),
        }


# Module-level convenience functions
_manager = None

def get_proxy_for_market(market: str) -> Optional[Dict]:
    """Get proxy configuration for a market."""
    global _manager
    if _manager is None:
        _manager = ProxyManager()
    return _manager.get_proxy(market)

from datetime import datetime

from flask_login import UserMixin
import bcrypt

db = SQLAlchemy()


class User(UserMixin, db.Model):
    """User account model."""
    __tablename__ = 'users'

    id = db.Column(db.Integer, primary_key=True)
    email = db.Column(db.String(255), unique=True, nullable=False)
    password_hash = db.Column(db.String(255), nullable=False)
    name = db.Column(db.String(100))
    preferred_currency = db.Column(db.String(3), default='USD')
    home_market = db.Column(db.String(2), default='US')
    xrp_wallet_address = db.Column(db.String(100))
    is_verified = db.Column(db.Boolean, default=False)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)

    def set_password(self, password):
        self.password_hash = bcrypt.hashpw(
            password.encode(), bcrypt.gensalt()
        ).decode()

    def check_password(self, password):
        return bcrypt.checkpw(
            password.encode(),
            self.password_hash.encode()
        )


class Deal(db.Model):
